import numpy as np

def fixedpt(f, x0, tol, Nmax):
    import numpy as np
    '''x0 = initial guess'''
    '''Nmax = max number of iterations'''
    '''tol = stopping tolerance'''

    count = 0
    phat = np.array([])
    while count < Nmax:
        count = count + 1
        x1 = f(x0)
        phat = np.append(phat, x1)
        if abs(x1 - x0) < tol:
            xstar = x1
            ier = 0
            return [xstar, ier, count, phat]

        x0 = x1
        
    xstar = x1
    ier = 1
    count = 0
    return [xstar, ier, count, phat]
    
x0 = 1
tol = 1e-8
Nmax = 100

    return x**2 - 1

def conv_order(phat, p):
    err1 = np.abs(phat[2] - p)
    err2 = np.abs(phat[1] - p)
    err3 = np.abs(phat[0] - p)

    alpha = np.log(np.abs(err1/err2)) / np.log(np.abs(err2/err3))
    return alpha

def steffensons(g, p0, tol, Nmax):

    count = 0
    phat = np.array([])
    while count < Nmax:
        count = count + 1

        b = g(p0)
        c = g(b)
        p1 = p0 - (((b - p0)**2) / (c - 2*b + p0))
        phat = np.append(phat, p1)
        if abs(p1 - p0) < tol:
            pstar = p0
            ier = 0
            return [pstar, ier, count, phat]

        p0 = p1

    pstar = p1
    ier = 1
    count = 0
    return [pstar, ier, count, phat]
